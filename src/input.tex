    \newcommand{\namespace}[1][Name:]{
        \noindent ##1 \rule{14em}{0.3mm}
    }
    \newcommand{\examtitle}[1]{
        {\noindent\fontsize{33}{39}\selectfont ##1}
    }
    \newcommand{\examyear}[1]{

\NewEnviron{examinfo}[1]{
    \setlength{\extrarowheight}{4pt}
    \par\addvspace{3ex}
    {\noindent\fontsize{12}{13}\selectfont
        \begin{tabular*}{\textwidth}{@{}p{6.75em}@{}p{0.5em}@{}p{\dimexpr\textwidth-7.5em\relax}}
            \cline{1-1}
            {\raggedright\textbf{#1}\par} & & \BODY
        \end{tabular*}}
    \par\addvspace{4ex}
}

}

}

\NewEnviron{multi}[1][1]{

    \end{tabular*}

}

\newcommand{\choice}[2][A]{
    \vspace{2ex}
    \noindent
    \noindent
    \begin{tabular*}{\textwidth}{@{}p{2.5em}@{}p{\dimexpr\textwidth-9.5em\relax}}
        #1. & #2
    \end{tabular*}
    \par
}

\newcommand{\writinglines}[1][1]{
    \vspace{2ex}
    \foreach \i in {1,...,#1} {
            \vspace{0.6cm}
            \dotfill
            \par
        }
}

\newcommand{\annotatedpagebreak}[1]{
    \vspace{2ex}

    \pagebreak
}
