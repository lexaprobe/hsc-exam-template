\NewEnviron{examinfo}[1]{
    \setlist[itemize]{leftmargin=*}
    \setlength{\extrarowheight}{4pt}
    \par\addvspace{3ex}
    {\fontsize{12}{13}\selectfont
        \begin{tabular*}{\textwidth}{@{}p{6.75em}@{}p{0.5em}@{}p{\dimexpr\textwidth-7.5em\relax}}
            \cline{1-1}
            {\raggedright\textbf{#1}\par} & & \BODY
        \end{tabular*}}
    \par\addvspace{4ex}
}

}

\NewEnviron{question}[2]{

    \subsection*{\textbf{Question #1} (#2 marks)}

    \newcommand{\continued}{
        \subsection*{Question #1 (continued)}
    }

    \newcommand{\questionbody}[2][]{
        \par\addvspace{3ex}
        \begin{tabular*}{\textwidth}{@{}p{\dimexpr\textwidth-3em\relax}@{\extracolsep{\fill}}r@{}}
            ##2 & \textbf{##1}
        \end{tabular*}
        \par\addvspace{4ex}
    }

    \newcommand{\questionpart}[3][]{
        \par\addvspace{3ex}
        \begin{tabular*}{\textwidth}{@{}p{2.5em}@{}p{\dimexpr\textwidth-6em\relax}@{\extracolsep{\fill}}r@{}}
            (##2) & ##3 & \textbf{##1}
        \end{tabular*}
        \par\addvspace{4ex}
    }

    \newcommand{\questionsubpart}[3][]{
        \par\addvspace{3ex}
        \begin{tabular*}{\textwidth}{@{}p{2.5em}@{}p{\dimexpr\textwidth-8.5em\relax}@{\hspace{3em}}r@{}}
            (##2) & ##3 & \textbf{##1}
        \end{tabular*}
        \par\addvspace{4ex}
    }

    \BODY
}

\NewEnviron{multi}[1]{

    \newcommand{\choice}[2][A]{
        \vspace{2ex}
        \begin{tabular*}{\textwidth}{@{}p{2.5em}@{}p{\dimexpr\textwidth-9.5em\relax}}
            ##1. & ##2
        \end{tabular*}
        \par
    }

    \par\addvspace{3ex}
    \begin{tabular*}{\textwidth}{@{}p{2.5em}@{}p{\dimexpr\textwidth-4em\relax}}
        \textbf{#1} & \BODY
    \end{tabular*}
    \par\addvspace{4ex}
}

\newcommand{\writinglines}[1][1]{
    \vspace{2ex}
    \foreach \i in {1,...,#1} {
            \vspace{0.6cm}
            \dotfill
            \par
        }
}

\newcommand{\annotatedpagebreak}[1]{
    \vspace{2ex}

    \pagebreak
