\usepackage{pgffor}
\usepackage{tikz}
\usepackage{array}
\usepackage[utf8]{inputenc}
\usepackage[margin=1in]{geometry}
\usepackage{fontspec}
\usepackage{titlesec}
\usepackage{environ}
\setmainfont{Times New Roman}

\titleformat{\section}
{\large\bfseries}
{\thesection}
{1em}
{}

\titleformat{\subsection}
{\normalfont\mdseries}
{\thesubsection}
{1em}
{}

\NewEnviron{sectionheader}[1]
{
    \section*{#1}
    \BODY
    \\
    \hrule
}

\newcommand{\questionheader}[2]{
    \subsection*{\textbf{#1} (#2)}
}

\newcommand{\questionsubheader}[2]{
    \subsection*{#1 (#2)}
}

\NewEnviron{question}[2]{
    \par\addvspace{1.5em}
    \noindent
    \begin{tabular*}{\textwidth}{@{}p{2.5em}@{}p{\dimexpr\textwidth-6em\relax}@{\extracolsep{\fill}}r@{}}
        #1 & \BODY & \textbf{#2}
    \end{tabular*}
    \par\addvspace{2em}
}

\NewEnviron{questionpart}[2]{
    \par\addvspace{1.5em}
    \noindent
    \begin{tabular*}{\textwidth}{@{}p{2.5em}@{}p{\dimexpr\textwidth-9.5em\relax}@{\hspace{4em}}r@{}}
        #1 & \BODY & \textbf{#2}
    \end{tabular*}
    \par\addvspace{2em}
}

\NewEnviron{multi}[1]{
    \par\addvspace{1.5em}
    \noindent
    \begin{tabular*}{\textwidth}{@{}p{2.5em}@{}p{\dimexpr\textwidth-4em\relax}}
        #1 & \BODY
    \end{tabular*}
    \par\addvspace{2em}
}

\newcommand{\choice}[2]{
    \vspace{1em}
    \noindent
    \noindent
    \begin{tabular*}{\textwidth}{@{}p{2.5em}@{}p{\dimexpr\textwidth-9.5em\relax}}
        #1 & #2
    \end{tabular*}
    \par
}

\newcommand{\writinglines}[1]{
    \vspace{1em}
    \foreach \i in {1,...,#1} {
            \vspace{1em}
            \dotfill
            \par
        }
}

\newcommand{\annotatedpagebreak}[1]{
    \vspace{1em}
    {\centering\textbf{#1}\par}
    \pagebreak
}
